**Goal:** Dejar la tesis completa en Markdown, con la estructura del docente, lista para copiar al

## Global Constraints

- Idioma de la tesis: español; citas IEEE numeradas `[n]` en todo el documento, sin mezcla.
- Hipótesis: ≥ 30 % de ahorro de gasto premium contra todo-premium y tasa de aceptables no inferior en más de 5 pp; **sin** mención del perfil de contexto (ni como desviación).
- Veredicto: **se verifica**; encuadre "aun midiendo de forma conservadora, supera el umbral". Nunca una cifra de ahorro "con juicio humano".
- En resumen y 12.2 no se usa "limitado" como titular; las amenazas van en §11.2.5 con redacción neutra.
- 4 OE sin meses (OE1 relevamiento y estándar de referencia; OE2 enrutamiento y consola operativa; OE3 medición de calidad y benchmarks; OE4 documentación y conclusiones).
- Cero gasto de API; no se generan datos.
- No modificar `scripts/metric_validation/`, `benchmarks/metric-validation/`, `docs/evaluation.md`, `tests/test_metric_validation.py` (leerlos e importarlos sí).
- No activar el router aprendido en `.env`.
- Suites pesadas con `darwin-throttle`; vitest `--maxWorkers=2`; nunca dos suites a la vez.
- Figuras y tablas: número + título + referencia en el texto ("Figura 10.1", "Tabla 11.3").
- Marcas abiertas permitidas al final: solo `> COMPLETAR (Joaquín): …` y `> CAPTURA (fase 7): …`.
- Commits convencionales, terminados en `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
# scripts/thesis/blocks.py
"""Find and rewrite the generated blocks of the thesis source.

A block is ``<!-- GEN:name -->`` + newline + content + newline + ``<!-- /GEN:name -->``.
Names are unique across every Markdown file of the thesis directory.
"""

from __future__ import annotations

import re
from pathlib import Path

OPEN = re.compile(r"<!-- GEN:([a-z0-9-]+) -->")


class BlockError(ValueError):
    pass


def _pattern(name: str) -> re.Pattern[str]:
    n = re.escape(name)
    return re.compile(rf"(<!-- GEN:{n} -->\n)(.*?)(\n<!-- /GEN:{n} -->)", re.DOTALL)


def find_blocks(root: Path) -> dict[str, Path]:
    found: dict[str, Path] = {}
    for path in sorted(root.glob("*.md")):
        for name in OPEN.findall(path.read_text(encoding="utf-8")):
            if name in found:
                raise BlockError(f"GEN block {name!r} appears in {found[name]} and {path}")
            found[name] = path
    return found


def replace_block(text: str, name: str, content: str) -> str:
    pattern = _pattern(name)
    if not pattern.search(text):
        raise BlockError(f"No GEN block {name!r} in the thesis source.")
    return pattern.sub(lambda m: m.group(1) + content + m.group(3), text, count=1)


def _current(text: str, name: str) -> str:
    match = _pattern(name).search(text)
    if not match:
        raise BlockError(f"GEN block {name!r} is not closed.")
    return match.group(2)


def apply(contents: dict[str, str], root: Path, *, write: bool) -> list[str]:
    located = find_blocks(root)
    missing, unknown = sorted(set(located) - set(contents)), sorted(set(contents) - set(located))
    if missing or unknown:
        raise BlockError(f"no generator for {missing}; no block in the thesis for {unknown}")
    stale: list[str] = []
    texts = {path: path.read_text(encoding="utf-8") for path in set(located.values())}
    for name in sorted(contents):
        path = located[name]
        if _current(texts[path], name) != contents[name]:
            stale.append(name)
            texts[path] = replace_block(texts[path], name, contents[name])
    if write:
        for path in {located[n] for n in stale}:
            path.write_text(texts[path], encoding="utf-8")
    return stale


def update_block(name: str, content: str, root: Path) -> None:
    located = find_blocks(root)
    if name not in located:
        raise BlockError(f"No GEN block {name!r} in {root}.")
    path = located[name]
    path.write_text(replace_block(path.read_text(encoding="utf-8"), name, content), encoding="utf-8")
```

```python
"""Regenerate every generated block of the thesis from the committed reports.

    python -m scripts.thesis.tables [--check]

Reads only what is versioned under ``benchmarks/``: no network, no training.
``--check`` writes nothing and exits 1 when a block is stale.
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

from scripts.metric_validation import rubric
from scripts.thesis import blocks

THESIS_DIR = Path("docs/tfc/tesis")
GT_ROOT = Path("benchmarks/ground-truth/v1")
ROUTER = Path("benchmarks/router/v1")
RESULTS = Path("benchmarks/results")
METRIC = Path("benchmarks/metric-validation")
BASELINE_RUNS = ("20260819T225557Z", "20260819T225703Z", "20260819T225713Z")
POLICY_NAMES = {
    "all_local": "todo local (qwen2.5:7b)",
    "all_economy": "todo económico (claude-haiku-4.5)",
    "all_frontier": "todo frontier (gpt-4.1)",
    "oracle": "oráculo",
    "heuristic_premium_frontier": "heurística de dos reglas → frontier",
    "heuristic_premium_economy": "heurística de dos reglas → económico",
}


def _json(path: Path) -> dict:
    if not path.exists():
        raise FileNotFoundError(f"missing report {path}")
    return json.loads(path.read_text(encoding="utf-8"))


def _usd(cost_per_prompt: float) -> str:
    return f"{cost_per_prompt * 1000:.2f}"


def baseline_savings(runs_dir: Path) -> str:
    parts, served = [], set()
    for run_id in BASELINE_RUNS:
        s = _json(runs_dir / run_id / "report.json")["summary"]
        avoided, spent = s["estimated_premium_cost_avoided"], s["estimated_premium_cost"]
        parts.append(f"{avoided / (spent + avoided):.1%}".replace("%", " %") + f" (corrida {run_id})")
        routes = s["route_distribution"]
        served.add((s["passed"], s["total_requests"], routes.get("local", 0) + routes.get("cache", 0)))
    passed, total, cheap = sorted(served)[0]
    return (f"Línea de base (heurística de dos reglas, {total} escenarios, 2026-08-19): "
            f"{', '.join(parts)} de gasto premium evitado; en cada corrida pasaron "
            f"{passed}/{total} escenarios y {cheap} de {total} pedidos se sirvieron por el modelo "
            f"local o el caché. Estas corridas no miden la calidad de lo servido.")


def metric_validation(root: Path) -> str:
    r = _json(root / "report.json")
    chosen = r["chosen_prefix"]
    natural = next(p for p in r["prefix_sweep_natural"] if p["prefix"] == chosen)
    ci = natural["auc_ci"]
    scale = {s["kind"]: s for s in r["scale_by_kind"]}
    ref = r["reference_rater"]
    labels = {}
    for rater in r["raters"]:
        path = root / "labels" / f"{rater}.jsonl"
        if not path.exists():
            raise FileNotFoundError(f"missing labels {path}")
        rows = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line]
        labels[rater] = {row["pair_id"]: row["grade"] for row in rows}
    shared = sorted(labels[ref])
    raters_json = _json(root / "raters.json")
    rates = []
    for rater in r["raters"]:
        subs = sum(rubric.is_substitutable(labels[rater][p]) for p in shared if p in labels[rater])
        model = (raters_json[rater]["note"].split(",")[0] if raters_json[rater]["kind"] != "human"
                 else "lector humano")
        rates.append(f"{model} {subs / len(shared):.0%}".replace("%", " %"))
    return (f"Sobre {natural['pairs']} pares en inglés con nota humana, la similitud coseno del "
            f"embedding (prefijo `{chosen}`) separó las respuestas sustituibles de las que no con "
            f"AUC {natural['auc']:.2f} (IC 95 % {ci['low']:.2f}–{ci['high']:.2f}), peor que el azar. "
            f"La mediana del coseno entre respuesta local y premium fue "
            f"{scale['local_vs_premium']['median']:.3f}, prácticamente igual a la de dos modelos "
            f"premium entre sí ({scale['premium_vs_premium']['median']:.3f}): la métrica se satura y "
            f"no resuelve calidad en ese rango. En esos mismos pares, la proporción juzgada "
            f"sustituible fue: {', '.join(rates)}.")


def router_fixed_policies(report: dict) -> str:
    rows = ["| Política | Costo (USD / 1000 prompts) | Calidad |", "|---|---|---|"]
    for key, label in POLICY_NAMES.items():
        p = report["baselines"][key]
        rows.append(f"| {label} | {_usd(p['cost'])} | {p['quality']:.3f} |")
    n = report["nested"]["0.95"]
    rows.append(f"| router aprendido, objetivo 0.95 (anidado) | {_usd(n['cost'])} | {n['quality']:.3f} |")
    return "\n".join(rows)


def router_sensitivity(report: dict) -> str:
    levels = sorted(report["learned"]["cost_at"], key=float)
    rows = ["| Regla de etiquetado | " + " | ".join(f"calidad ≥ {q}" for q in levels)
            + " | todo frontier |", "|---|" + "---|" * (len(levels) + 1)]
    names = {"tiers.R1_unanimous.jsonl": "R1 unanimidad", "tiers.R2_majority.jsonl": "R2 mayoría"}
    for key, v in report["sensitivity"].items():
        rows.append(f"| {names.get(key, key)} | " + " | ".join(_usd(v["cost_at"][q]) for q in levels)
                    + f" | {_usd(v['all_frontier_cost'])} |")
    main = report["learned"]["cost_at"]
    rows.append("| R3 media ordinal (elegida) | " + " | ".join(_usd(main[q]) for q in levels)
                + f" | {_usd(report['baselines']['all_frontier']['cost'])} |")
    return "\n".join(rows)


def router_latency(latency: dict) -> str:
    rows = ["| Modelo | Mediana (s) | p90 (s) | n |", "|---|---|---|---|"]
    for v in sorted(latency.values(), key=lambda v: v["median_s"]):
        rows.append(f"| {v['model']} | {v['median_s']:.1f} | {v['p90_s']:.1f} | {v['n']} |")
    return "\n".join(rows)


def router_frontier_points(report: dict) -> str:
    wanted = (0.80, 0.85, 0.90, 0.95, 0.98)
    pareto = report["learned"]["pareto"]
    rows = ["| Calidad mínima | τ local | τ económico | Costo (USD / 1000) | Local | Económico | Frontier |",
            "|---|---|---|---|---|---|---|"]
    for q in wanted:
        p = min((p for p in pareto if p["quality"] >= q), key=lambda p: p["cost"])
        s = p["share"]
        rows.append(f"| {q:.2f} | {p['tau_local']:.2f} | {p['tau_economy']:.2f} | {_usd(p['cost'])} | "
                    f"{s['local']:.0%} | {s['economy']:.0%} | {s['frontier']:.0%} |")
    return "\n".join(rows)


def render_all() -> dict[str, str]:
    from scripts.ground_truth import report as gt_report
    from scripts.router import train

    summary = gt_report.summarise(GT_ROOT)
    router = _json(ROUTER / "report.json")
    latency = _json(ROUTER / "latency.json")
    return {
        "corpus-fase2": gt_report.thesis_corpus(summary),
        "judges-fase2": gt_report.thesis_judges(summary),
        "router-fase3": train.thesis_block(router),
        "baseline-savings": baseline_savings(RESULTS),
        "metric-validation": metric_validation(METRIC),
        "router-fixed-policies": router_fixed_policies(router),
        "router-sensitivity": router_sensitivity(router),
        "router-latency": router_latency(latency),
        "router-frontier-points": router_frontier_points(router),
    }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--check", action="store_true", help="write nothing; exit 1 if stale")
    args = parser.parse_args(argv)
    try:
        stale = blocks.apply(render_all(), THESIS_DIR, write=not args.check)
    except (FileNotFoundError, blocks.BlockError, KeyError) as exc:
        print(f"thesis-tables: {exc}", file=sys.stderr)
        return 2
    if args.check and stale:
        print("stale GEN blocks: " + ", ".join(stale), file=sys.stderr)
        return 1
    print(("updated: " + ", ".join(stale)) if stale else "thesis tables up to date")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
